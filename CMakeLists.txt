cmake_minimum_required(VERSION 3.20)
project(video_player LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version" FORCE)
endif()
set(NLE_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nle-sdk")
if(NOT EXISTS "${NLE_SDK_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Initialize the SDK first: git submodule update --init --recursive")
endif()
# Only one rendering implementation and one set of third-party dependencies: nle-sdk.
set(BUILD_TESTING OFF CACHE BOOL "Build SDK tests separately" FORCE)
set(NLE_SDK_BUILD_DEMOS OFF CACHE BOOL "Build SDK demos" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Link SDK into the addon" FORCE)
set(NLE_SDK_PLUGIN_DIR "${CMAKE_BINARY_DIR}/plugins" CACHE PATH "Codec plugin directory" FORCE)
add_subdirectory(third_party/nle-sdk EXCLUDE_FROM_ALL)
execute_process(COMMAND node -p "require('node-addon-api').include"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE NAPI_INCLUDE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
string(REPLACE "\"" "" NAPI_INCLUDE "${NAPI_INCLUDE}")
execute_process(COMMAND node -p "require('node-api-headers').include_dir"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE NODE_INCLUDE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
add_library(video_player MODULE
    src/addon.cpp src/addon/root_wrap.cpp src/addon/layer_wrap.cpp src/addon/group_layer_wrap.cpp)
target_include_directories(video_player PRIVATE "${NAPI_INCLUDE}" "${NODE_INCLUDE}")
target_link_libraries(video_player PRIVATE nle_sdk::nle_sdk)
target_compile_definitions(video_player PRIVATE NAPI_VERSION=8 NAPI_CPP_EXCEPTIONS)
set_target_properties(video_player PROPERTIES PREFIX "" SUFFIX ".node" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(APPLE)
    target_compile_definitions(video_player PRIVATE NLE_PLUGIN_NAME="media_codec_apple")
    target_link_options(video_player PRIVATE "-undefined" "dynamic_lookup")
    add_dependencies(video_player nle_media_codec_apple_plugin)
else()
    target_compile_definitions(video_player PRIVATE NLE_PLUGIN_NAME="media_codec_ffmpeg")
    add_dependencies(video_player nle_media_codec_ffmpeg_plugin)
endif()

# Only runtime artifacts are published; intermediate objects/static libraries stay in build/.
if(APPLE)
    set_target_properties(video_player PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@loader_path/lib")
    set_target_properties(nle_media_codec_apple_plugin PROPERTIES
        BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@loader_path/../../lib")

endif()
